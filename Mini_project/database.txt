-- 1. Create the Database

CREATE DATABASE IF NOT EXISTS the_wellington;

USE the_wellington;

-- 2. Create Users Table (Stores both Staff and Customers)

CREATE TABLE users (

    user_id INT AUTO_INCREMENT PRIMARY KEY,

    username VARCHAR(50) NOT NULL UNIQUE,

    password VARCHAR(255) NOT NULL, -- In real app, use password_hash()

    full_name VARCHAR(100) NOT NULL,

    email VARCHAR(100),

    role ENUM('staff', 'customer') DEFAULT 'customer',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

);

-- 3. Create Restaurant Tables (Physical tables in the room)

CREATE TABLE restaurant_tables (

    table_id INT AUTO_INCREMENT PRIMARY KEY,

    table_number VARCHAR(10) NOT NULL,

    capacity INT NOT NULL, -- e.g., 2 seater, 4 seater

    status ENUM('available', 'occupied', 'reserved') DEFAULT 'available'

);

-- 4. Create Menu Table

CREATE TABLE menu (

    item_id INT AUTO_INCREMENT PRIMARY KEY,

    name VARCHAR(100) NOT NULL,

    description TEXT,

    price DECIMAL(10, 2) NOT NULL,

    category VARCHAR(50) -- e.g., 'Main Course', 'Starter'

);

-- 5. Create Reservations Table

CREATE TABLE reservations (

    reservation_id INT AUTO_INCREMENT PRIMARY KEY,

    user_id INT, -- Links to the Customer

    table_id INT, -- Links to the Physical Table

    reservation_date DATE NOT NULL,

    reservation_time TIME NOT NULL,

    guests INT NOT NULL,

    status ENUM('pending', 'confirmed', 'completed', 'cancelled') DEFAULT 'pending',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id) REFERENCES users(user_id),

    FOREIGN KEY (table_id) REFERENCES restaurant_tables(table_id)

);

-- 6. Create Reservation Orders (The Pre-ordered Food)

CREATE TABLE reservation_orders (

    order_id INT AUTO_INCREMENT PRIMARY KEY,

    reservation_id INT,

    item_id INT, -- Links to Menu

    quantity INT DEFAULT 1,

    FOREIGN KEY (reservation_id) REFERENCES reservations(reservation_id) ON DELETE CASCADE,

    FOREIGN KEY (item_id) REFERENCES menu(item_id)

);
